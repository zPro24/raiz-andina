function renderizarCarrito() { const contenedor = document.getElementById('items-carrito-container'); const totalElemento = document.getElementById('cart-total-precio'); const carrito = obtenerCarrito(); if (!contenedor) return; if (carrito.length === 0) { contenedor.innerHTML = `

Tu cesta está vacía 🌾

`; if (totalElemento) totalElemento.textContent = '$0'; return; } let totalAcumulado = 0; contenedor.innerHTML = carrito.map(item => { const subtotal = item.precio * item.cantidad; totalAcumulado += subtotal; return `

${item.nombre}

$${item.precio.toLocaleString('es-CO')} c/u
${item.cantidad}
`; }).join(''); if (totalElemento) { totalElemento.textContent = `$${totalAcumulado.toLocaleString('es-CO')}`; } }
